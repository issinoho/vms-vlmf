# PTYDRIVE script for test_display.com: the highlight bar in colour and MONO.
# Read-only: it only moves the cursor.
2 dcl | set terminal/width=80/page=24\r
8 colour | vlmf [.live]view.ldb\r
2 colour down | \e[B
2 colour quit | \e[21~
2 mono define | define/process vlmf$display mono\r
8 mono | vlmf [.live]view.ldb\r
2 mono down | \e[B
2 mono quit | \e[21~
2 dcl after | deassign/process vlmf$display\r
